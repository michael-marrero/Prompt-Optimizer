    db = app.state.db

    # Pre-stream errors → HTTPException (D-08)
    thread = await get_thread(db, thread_id)
    if thread is None:
        raise HTTPException(status_code=404, detail="thread not found")
    history_rows = await get_thread_messages(db, thread_id)

    # Convert DB Message rows to Phase 1 history shape (Phase 1 decide() expects list of dicts)
    history = [{"role": m.role, "content": m.text} for m in history_rows]
    # Adapter consumes list of Phase 2 AdapterMessage (different name, same shape)
    adapter_history = [AdapterMessage(role=m.role, content=m.text) for m in history_rows]

    turn_id = secrets.token_urlsafe(12)
    logger.info("turn_start thread_id=%s user_msg_len=%d turn_id=%s", thread_id, len(body.message), turn_id)

    # Routing decision: synthesise on override, else call decide() in thread pool (D-16 / API-07)
    if body.override_backend:
        decision = _synthesize_override_decision(body.override_backend)
    else:
        decision = await asyncio.to_thread(
            decide, body.message, history, app.state.artifacts, app.state.settings
        )

    logger.info(
        "routing_decision backend=%s model=%s rationale='%s' confidence=%.3f turn_id=%s",
        decision.backend, decision.model_or_agent, decision.rationale, decision.confidence, turn_id,
    )

    # STORE-06 + D-05: append jsonl BEFORE adapter dispatch
    await append_routing_decisions_jsonl(decision, thread_id, turn_id)

    # D-12 STRICT AND gate + D-15 lazy adapter (HTTPException 400 if computer_use opt-out)
    adapter = await _get_or_create_adapter(app, decision.backend)

    max_cost_usd = body.max_cost_usd or app.state.settings.get("default_max_cost_usd") or DEFAULT_PER_TURN_COST_USD
    options = AdapterOptions(
        model=decision.model_or_agent,
        max_cost_usd=max_cost_usd,
        max_steps=None,  # adapter applies its own default cap (25 claude_code / 15 computer_use)
        cwd=None,        # Wave 5 will add per-thread workspace for claude_code; Phase 2 default tmpdir is fine for v1
        routing_signals=decision.signals,
    )

    # SSE generator — PATTERNS Excerpt 2 lift-source
    async def event_stream():
        buffer: list[ChatChunk] = []
        start_t = asyncio.get_event_loop().time()
        try:
            async for chunk in adapter.stream(body.message, adapter_history, options):
                # Wave 5 will WRAP this generator with a Screenshot blob-externalize step (STORE-04).
                # Wave 4 ships pass-through; Wave 5 inserts the maybe_externalize call here.
                buffer.append(chunk)
                yield ServerSentEvent(
                    event=chunk.type,
                    data=chunk.model_dump_json(),
                )
                if isinstance(chunk, Done):
                    break
                # Proactive disconnect check (defense-in-depth; sse-starlette
                # _listen_for_disconnect also auto-cancels under real network — but
                # under ASGITransport this poll is a no-op per Pitfall 6).
                if await request.is_disconnected():
                    break
        except asyncio.CancelledError:

            # the upstream adapter's CancelledError handler fires (which closes
            # the provider connection within the 2s budget per BACKEND-07 / API-06).
            err = StreamError(code="cancelled", message="Stream cancelled by caller.", retriable=True)
            yield ServerSentEvent(event=err.type, data=err.model_dump_json())
            buffer.append(err)
            done = Done(routing_signals=decision.signals)
            yield ServerSentEvent(event=done.type, data=done.model_dump_json())
            buffer.append(done)
            raise
        finally:
            # STORE-05 + D-04: ONE BEGIN/COMMIT on Done.
            # The Phase 2 D-04 terminal-Done invariant guarantees the buffer's
            # last chunk is a Done at this point (happy path; cancellation; or
            # any provider-error path that the adapter handled with StreamError+Done).
            if buffer and isinstance(buffer[-1], Done):
                # Determine status from the buffer's last StreamError (if any).
                last_error = next((c for c in buffer if isinstance(c, StreamError)), None)
                if last_error is None:
                    status = "complete"
                elif last_error.code == "cancelled":
                    status = "cancelled"
                else:
                    status = "error"
                user_message_id = secrets.token_urlsafe(12)
                assistant_message_id = secrets.token_urlsafe(12)
                routing_decision_id = secrets.token_urlsafe(12)
                try:
                    await persist_turn(
                        db,
                        thread_id=thread_id,
                        user_text=body.message,
                        user_message_id=user_message_id,
                        assistant_message_id=assistant_message_id,
                        routing_decision_id=routing_decision_id,
                        buffer=buffer,
                        decision=decision,
                        status=status,
                    )
                except Exception:
                    logger.exception("persist_turn failed for turn_id=%s", turn_id)
                # D-19 terminal log
                done_chunk = buffer[-1]
                logger.info(
                    "turn_done thread_id=%s backend=%s cost_usd=%s tokens_in=%s tokens_out=%s latency_ms=%s status=%s turn_id=%s",
                    thread_id, decision.backend,
                    getattr(done_chunk, "cost_usd", None),
                    getattr(done_chunk, "tokens_in", None),
                    getattr(done_chunk, "tokens_out", None),
                    getattr(done_chunk, "latency_ms", None) or int((asyncio.get_event_loop().time() - start_t) * 1000),
                    status, turn_id,
                )

    return EventSourceResponse(event_stream(), ping=15)
