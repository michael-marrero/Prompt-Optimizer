    5. Helper function _predict_with_calibrated(prompt: str, raw_text_for_vectorizer: pd.Series, artifact: dict, extractor: PromptFeatureExtractor, extra_values: dict | None = None) -> tuple[str, float, list[tuple[str, float]]]:
       - text_features = artifact["vectorizer"].transform(raw_text_for_vectorizer)
       - numeric_df = _build_numeric_features(prompt, artifact["feature_columns"], extractor, extra_values)
       - numeric_scaled = artifact["scaler"].transform(numeric_df)
       - combined = hstack([text_features, csr_matrix(numeric_scaled)])
       - pred_encoded = artifact["model"].predict(combined)[0]
       - pred_label = artifact["label_encoder"].inverse_transform([pred_encoded])[0]
       - probs = artifact["model"].predict_proba(combined)[0]
       - confidence_table = sorted(zip(artifact["label_encoder"].classes_, probs), key=lambda kv: kv[1], reverse=True)
       - return pred_label, confidence_table[0][1], list(confidence_table)
