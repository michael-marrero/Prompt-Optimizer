ECE helper (RESEARCH lines 884-901; 16 lines; no new dep):

    from sklearn.calibration import calibration_curve
    import numpy as np

    def expected_calibration_error(y_true_binary, y_prob_max, n_bins=10):
        edges = np.linspace(0.0, 1.0, n_bins + 1)
        n = len(y_true_binary)
        ece = 0.0
        for lo, hi in zip(edges[:-1], edges[1:]):
            in_bin = (y_prob_max > lo) & (y_prob_max <= hi)
            if in_bin.sum() == 0:
                continue
            acc = y_true_binary[in_bin].mean()
            conf = y_prob_max[in_bin].mean()
            ece += (in_bin.sum() / n) * abs(acc - conf)
        return float(ece)

For multiclass, y_true_binary is (y_pred == y_test).astype(int) — the "did the argmax win?" view; y_prob_max is probs.max(axis=1).
